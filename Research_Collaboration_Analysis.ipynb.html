    "Researcher": list(G_largest.nodes()),
    "Degree Centrality": [
        degree_centrality[node] for node in G_largest.nodes()
    ],
    "Closeness Centrality": [
        closeness_centrality[node] for node in G_largest.nodes()
    ],
    "Betweenness Centrality": [
        betweenness_centrality[node] for node in G_largest.nodes()
    ]
})

# Display top researchers based on average centrality

    ["Degree Centrality",
     "Closeness Centrality",
     "Betweenness Centrality"]
].mean(axis=1)

top_overall = results.sort_values(
    "Overall Centrality",
    ascending=False
).head(10)
